find_program(PVPYTHON_EXECUTABLE NAMES pvpython REQUIRED)

set(THESIS_PARAVIEW_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

file(GLOB THESIS_PARAVIEW_SCRIPTS
  CONFIGURE_DEPENDS
  "${THESIS_PARAVIEW_SOURCE_DIR}/*/generate.py"
)

set(THESIS_PARAVIEW_GENERATED_FIGURES "")
set(THESIS_PARAVIEW_INPUTS "")

foreach(PARAVIEW_SCRIPT IN LISTS THESIS_PARAVIEW_SCRIPTS)
  get_filename_component(FIGURE_DIR "${PARAVIEW_SCRIPT}" DIRECTORY)
  get_filename_component(FIGURE_NAME "${FIGURE_DIR}" NAME)

  set(BINARY_STAMP "${CMAKE_CURRENT_BINARY_DIR}/${FIGURE_NAME}.stamp")
  set(SOURCE_STAMP "${THESIS_SOURCE_RASTER_DIR}/${FIGURE_NAME}.stamp")
  set(PARAVIEW_OUTPUTS
    "${BINARY_STAMP}"
    "${SOURCE_STAMP}"
  )

  file(GLOB_RECURSE PARAVIEW_CASE_INPUTS
    CONFIGURE_DEPENDS
    "${FIGURE_DIR}/case/*"
  )

  add_custom_command(
    OUTPUT ${PARAVIEW_OUTPUTS}
    COMMAND "${CMAKE_COMMAND}" -E make_directory
      "${THESIS_BINARY_RASTER_DIR}"
      "${THESIS_SOURCE_RASTER_DIR}"
    COMMAND "${PVPYTHON_EXECUTABLE}"
      "${PARAVIEW_SCRIPT}"
      "--output-dir=${THESIS_BINARY_RASTER_DIR}"
      "--output-dir=${THESIS_SOURCE_RASTER_DIR}"
      "--output-prefix=${FIGURE_NAME}"
    COMMAND "${CMAKE_COMMAND}" -E touch
      "${BINARY_STAMP}"
      "${SOURCE_STAMP}"
    DEPENDS
      "${PARAVIEW_SCRIPT}"
      ${PARAVIEW_CASE_INPUTS}
    COMMENT "Generating ParaView figure ${FIGURE_NAME}"
    VERBATIM
  )

  list(APPEND THESIS_PARAVIEW_INPUTS
    "${PARAVIEW_SCRIPT}"
    ${PARAVIEW_CASE_INPUTS}
  )
  list(APPEND THESIS_PARAVIEW_GENERATED_FIGURES
    ${PARAVIEW_OUTPUTS}
  )
endforeach()

add_custom_target(thesis_paraview_figures
  DEPENDS ${THESIS_PARAVIEW_GENERATED_FIGURES}
)

set(THESIS_PARAVIEW_GENERATED_FIGURES
  ${THESIS_PARAVIEW_GENERATED_FIGURES}
  PARENT_SCOPE
)
set(THESIS_PARAVIEW_INPUTS
  ${THESIS_PARAVIEW_INPUTS}
  PARENT_SCOPE
)
