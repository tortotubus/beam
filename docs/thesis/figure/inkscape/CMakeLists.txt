find_program(INKSCAPE_EXECUTABLE NAMES inkscape REQUIRED)

set(THESIS_INKSCAPE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

file(GLOB THESIS_INKSCAPE_SVGS
  CONFIGURE_DEPENDS
  "${THESIS_INKSCAPE_SOURCE_DIR}/*.svg"
)

set(THESIS_INKSCAPE_GENERATED_FIGURES "")
set(THESIS_INKSCAPE_INPUTS "")

foreach(INKSCAPE_SVG IN LISTS THESIS_INKSCAPE_SVGS)
  get_filename_component(FIGURE_NAME "${INKSCAPE_SVG}" NAME_WE)

  set(SOURCE_TEX "${THESIS_SOURCE_VECTOR_DIR}/${FIGURE_NAME}.pdf_tex")
  set(SOURCE_PDF "${THESIS_SOURCE_VECTOR_DIR}/${FIGURE_NAME}.pdf")
  set(BINARY_TEX "${THESIS_BINARY_VECTOR_DIR}/${FIGURE_NAME}.pdf_tex")
  set(BINARY_PDF "${THESIS_BINARY_VECTOR_DIR}/${FIGURE_NAME}.pdf")

  add_custom_command(
    OUTPUT
      "${BINARY_TEX}"
      "${BINARY_PDF}"
      "${SOURCE_TEX}"
      "${SOURCE_PDF}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory
      "${THESIS_BINARY_VECTOR_DIR}"
      "${THESIS_SOURCE_VECTOR_DIR}"
    COMMAND "${INKSCAPE_EXECUTABLE}"
      "${INKSCAPE_SVG}"
      "--export-filename=${BINARY_PDF}"
      --export-latex
    COMMAND "${INKSCAPE_EXECUTABLE}"
      "${INKSCAPE_SVG}"
      "--export-filename=${SOURCE_PDF}"
      --export-latex
    DEPENDS "${INKSCAPE_SVG}"
    COMMENT "Generating Inkscape figure ${FIGURE_NAME}"
    VERBATIM
  )

  list(APPEND THESIS_INKSCAPE_INPUTS
    "${INKSCAPE_SVG}"
  )
  list(APPEND THESIS_INKSCAPE_GENERATED_FIGURES
    "${BINARY_TEX}"
    "${BINARY_PDF}"
    "${SOURCE_TEX}"
    "${SOURCE_PDF}"
  )
endforeach()

add_custom_target(thesis_inkscape_figures
  DEPENDS ${THESIS_INKSCAPE_GENERATED_FIGURES}
)

set(THESIS_INKSCAPE_GENERATED_FIGURES
  ${THESIS_INKSCAPE_GENERATED_FIGURES}
  PARENT_SCOPE
)
set(THESIS_INKSCAPE_INPUTS
  ${THESIS_INKSCAPE_INPUTS}
  PARENT_SCOPE
)
