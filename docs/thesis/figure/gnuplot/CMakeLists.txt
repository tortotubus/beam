find_program(GNUPLOT_EXECUTABLE NAMES gnuplot REQUIRED)

set(THESIS_GNUPLOT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

file(GLOB THESIS_GNUPLOT_SCRIPTS
  CONFIGURE_DEPENDS
  "${THESIS_GNUPLOT_SOURCE_DIR}/*.gp"
)

set(THESIS_GNUPLOT_GENERATED_FIGURES "")
set(THESIS_GNUPLOT_INPUTS "")

foreach(GNUPLOT_SCRIPT IN LISTS THESIS_GNUPLOT_SCRIPTS)
  get_filename_component(FIGURE_NAME "${GNUPLOT_SCRIPT}" NAME_WE)

  set(SOURCE_TEX "${THESIS_SOURCE_VECTOR_DIR}/${FIGURE_NAME}.tex")
  set(SOURCE_PDF "${THESIS_SOURCE_VECTOR_DIR}/${FIGURE_NAME}.pdf")
  set(BINARY_TEX "${THESIS_BINARY_VECTOR_DIR}/${FIGURE_NAME}.tex")
  set(BINARY_PDF "${THESIS_BINARY_VECTOR_DIR}/${FIGURE_NAME}.pdf")

  set(GNUPLOT_INPUTS "${GNUPLOT_SCRIPT}")
  set(GNUPLOT_ARGS
    -e "plot_output='${FIGURE_NAME}.tex'"
    -e "gnuplot_source_dir='${THESIS_GNUPLOT_SOURCE_DIR}'"
  )

  set(GNUPLOT_CSV "${THESIS_GNUPLOT_SOURCE_DIR}/${FIGURE_NAME}.csv")
  if(EXISTS "${GNUPLOT_CSV}")
    list(APPEND GNUPLOT_INPUTS "${GNUPLOT_CSV}")
    list(APPEND GNUPLOT_ARGS -e "csv_file='${GNUPLOT_CSV}'")
  endif()
  set(GNUPLOT_HUANG_CSV "${THESIS_GNUPLOT_SOURCE_DIR}/huang/fig13.csv")
  if(FIGURE_NAME STREQUAL "huang-tip-Re-200-KB-10e-4" AND EXISTS "${GNUPLOT_HUANG_CSV}")
    list(APPEND GNUPLOT_INPUTS "${GNUPLOT_HUANG_CSV}")
    list(APPEND GNUPLOT_ARGS -e "huang_csv='${GNUPLOT_HUANG_CSV}'")
  endif()
  list(APPEND THESIS_GNUPLOT_INPUTS ${GNUPLOT_INPUTS})

  add_custom_command(
    OUTPUT
      "${BINARY_TEX}"
      "${BINARY_PDF}"
      "${SOURCE_TEX}"
      "${SOURCE_PDF}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory
      "${THESIS_BINARY_VECTOR_DIR}"
      "${THESIS_SOURCE_VECTOR_DIR}"
    COMMAND "${CMAKE_COMMAND}" -E chdir "${THESIS_BINARY_VECTOR_DIR}"
      "${GNUPLOT_EXECUTABLE}"
      ${GNUPLOT_ARGS}
      "${GNUPLOT_SCRIPT}"
    COMMAND "${CMAKE_COMMAND}" -E chdir "${THESIS_SOURCE_VECTOR_DIR}"
      "${GNUPLOT_EXECUTABLE}"
      ${GNUPLOT_ARGS}
      "${GNUPLOT_SCRIPT}"
    DEPENDS ${GNUPLOT_INPUTS}
    COMMENT "Generating gnuplot figure ${FIGURE_NAME}"
    VERBATIM
  )

  list(APPEND THESIS_GNUPLOT_GENERATED_FIGURES
    "${BINARY_TEX}"
    "${BINARY_PDF}"
    "${SOURCE_TEX}"
    "${SOURCE_PDF}"
  )
endforeach()

add_custom_target(thesis_gnuplot_figures
  DEPENDS ${THESIS_GNUPLOT_GENERATED_FIGURES}
)

set(THESIS_GNUPLOT_GENERATED_FIGURES
  ${THESIS_GNUPLOT_GENERATED_FIGURES}
  PARENT_SCOPE
)
set(THESIS_GNUPLOT_INPUTS
  ${THESIS_GNUPLOT_INPUTS}
  PARENT_SCOPE
)
